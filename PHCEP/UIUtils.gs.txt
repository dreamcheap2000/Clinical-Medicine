function handleCustomEdit(e) {
  if (!e || !e.range) return;
  const sheetName = e.range.getSheet().getName();
  if (sheetName === 'LINE對話' || sheetName === '工作表1') {
    handleDoctorReplyEdit(e);
  } else if (sheetName === 'QA知識庫') {
    handleQARowEdit(e);
  }
}

function refreshSelectedRowSuggestions() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getActiveSheet();
  if (!sheet || !['LINE對話', '工作表1'].includes(sheet.getName())) {
    sheet = ss.getSheetByName('LINE對話') || ss.getSheetByName('工作表1');
    if (!sheet) {
      ss.toast("請先切換至「LINE對話」工作表！", "提示", 4);
      return;
    }
  }
  const row = sheet.getActiveCell().getRow();
  if (row <= 1) {
    ss.toast("請選取第 2 列以後的病患訊息列！", "提示", 4);
    return;
  }
  const patientText = String(sheet.getRange(row, 6).getValue() || '').trim();
  if (!patientText) {
    ss.toast(`第 ${row} 列 F 欄無訊息內容。`, "提示", 3);
    return;
  }
  suggestRepliesForUserMessage(patientText, row, 0.25);
  ss.toast(`第 ${row} 列 QA 建議回覆選單已產生！`, '完成', 3);
}

function formatCellDisplayForReply(content) {
  const text = String(content || '').trim();
  if (!text) return { isFormula: false, value: '' };
  const driveMatch = text.match(/\/file\/d\/([a-zA-Z0-9_-]+)/) || text.match(/[?&]id=([a-zA-Z0-9_-]+)/);
  if (driveMatch && driveMatch[1]) {
    const fileId = driveMatch[1];
    try {
      const file = DriveApp.getFileById(fileId);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      const viewUrl = `https://drive.google.com/file/d/${fileId}/view?usp=sharing`;
      return { isFormula: true, value: `=HYPERLINK("${viewUrl}", "${file.getName()}")` };
    } catch (e) {}
  }
  const directImgMatch = text.match(/https?:\/\/[^\s]+?\.(png|jpe?g|gif|webp)(\?[^\s]*)?/i);
  if (directImgMatch) {
    return { isFormula: true, value: `=HYPERLINK("${directImgMatch[0]}", "檢視圖片")` };
  }
  if (text.startsWith('=')) {
    return { isFormula: true, value: text };
  }
  return { isFormula: false, value: text };
}

function handleDoctorReplyEdit(e) {
  if (!e || !e.range) return;
  const range = e.range;
  const sheet = range.getSheet();
  const sheetName = sheet.getName();
  if (sheetName !== 'LINE對話' && sheetName !== '工作表1') return;
  const row = range.getRow();
  if (row <= 1) return; 
  const col = range.getColumn();
  
  if (col === 8) {
    let selectedText = String(e.value || range.getValue() || '').trim();
    if (!selectedText || selectedText.startsWith('───')) {
      range.clearContent();
      return;
    }
    const cache = CacheService.getScriptCache();
    const cachedData = cache.get(`QA_REPLY_CACHE_${selectedText}`);
    
    let fullText = selectedText;
    let qaRowIdx = null;
    
    try {
      if (cachedData && cachedData.startsWith('{')) {
        const parsed = JSON.parse(cachedData);
        fullText = parsed.reply;
        qaRowIdx = parsed.rowIdx;
      } else if (cachedData) {
        fullText = cachedData;
      }
    } catch (err) {}
    
    if (qaRowIdx) {
      try {
        const qaSheet = sheet.getParent().getSheetByName('QA知識庫');
        if (qaSheet) {
          const qaData = qaSheet.getRange(2, 1, qaSheet.getLastRow() - 1, 15).getValues();
          const targetReply = String(qaData[qaRowIdx - 2][6] || '').trim();
          const targetUrl = String(qaData[qaRowIdx - 2][7] || '').trim();
          
          const rowCountCell = qaSheet.getRange(qaRowIdx, 14);
          rowCountCell.setValue((Number(rowCountCell.getValue()) || 0) + 1);
          for (let i = 0; i < qaData.length; i++) {
            const rowReply = String(qaData[i][6] || '').trim();
            const rowUrl = String(qaData[i][7] || '').trim();
            if ((targetReply && rowReply === targetReply) || (targetUrl && rowUrl === targetUrl)) {
              const globalCountCell = qaSheet.getRange(i + 2, 15);
              globalCountCell.setValue((Number(globalCountCell.getValue()) || 0) + 1);
            }
          }
        }
      } catch (err) {}
    }
    const targetCellI = sheet.getRange(row, 9);
    targetCellI.setValue(fullText);
    range.clearContent(); 
    SpreadsheetApp.flush();
    targetCellI.activate(); 
    return;
  }
  
  if (col === 9) {
    let messageToSend = String(range.getValue() || '').trim();
    if (!messageToSend) return;
    const lock = LockService.getScriptLock();
    if (!lock.tryLock(8000)) return;
    const ss = sheet.getParent();
    try {
      const userId = String(sheet.getRange(row, 3).getValue() || '').trim();
      if (!userId) return;
      
      pushMultipleMessagesToLineUser(userId, messageToSend);
      ss.toast('訊息已成功發送給病患！', '發送成功', 3);
      
      const cellFormat = formatCellDisplayForReply(messageToSend);
      
      const maxCol = Math.max(sheet.getLastColumn(), 10);
      const historyWidth = maxCol - 10 + 1;
      if (historyWidth > 1) {
        const currentVals = sheet.getRange(row, 10, 1, historyWidth - 1).getValues()[0];
        const currentForms = sheet.getRange(row, 10, 1, historyWidth - 1).getFormulas()[0];
        
        for (let k = 0; k < currentVals.length; k++) {
          const targetCell = sheet.getRange(row, 11 + k);
          if (currentForms[k]) {
            targetCell.setFormula(currentForms[k]);
          } else if (currentVals[k] !== '' && currentVals[k] !== null) {
            targetCell.setValue(currentVals[k]);
          }
        }
      }
      
      const cellJ = sheet.getRange(row, 10);
      if (cellFormat.isFormula) {
        cellJ.setFormula(cellFormat.value);
      } else {
        cellJ.setValue(cellFormat.value);
      }
      
      range.clearContent();
      SpreadsheetApp.flush();
      range.activate();
    } catch (err) {
      Logger.log('[handleDoctorReplyEdit Error] ' + err.toString());
    } finally {
      lock.releaseLock();
    }
  }
}


function suggestRepliesForUserMessage(userMessage, targetRow, threshold = 0.25) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const qaSheet = ss.getSheetByName('QA知識庫');
  if (!qaSheet) return;

  const qaData = qaSheet.getRange(2, 1, qaSheet.getLastRow() - 1, 15).getValues();
  if (qaData.length === 0) return;

  const apiKey = getGeminiApiKey();
  const userVec = getGeminiEmbedding(userMessage, apiKey);
  if (!userVec) return;

  const suggestions = [];
  const cache = CacheService.getScriptCache();

  for (let i = 0; i < qaData.length; i++) {
    const reply = String(qaData[i][6] || '').trim();
    const replyVecCell = qaData[i][11];
    const replyVec = loadVectorFromDriveCell(replyVecCell);

    if (reply && replyVec) {
      const sim = cosineSimilarity(userVec, replyVec);
      if (sim >= threshold) {
        suggestions.push(reply);
        cache.put(`QA_REPLY_CACHE_${reply}`, JSON.stringify({ reply, rowIdx: i + 2 }), 3600);
      }
    }
  }

  const sheet = ss.getSheetByName('LINE對話');
  if (sheet && targetRow > 1) {
    const dropdownRange = sheet.getRange(targetRow, 8);
    if (suggestions.length > 0) {
      const rule = SpreadsheetApp.newDataValidation()
        .requireValueInList(suggestions, true)
        .setAllowInvalid(false)
        .build();
      dropdownRange.setDataValidation(rule);
    } else {
      dropdownRange.clearDataValidations();
    }
  }
}

function handleQARowEdit(e) {
  if (!e || !e.range) return;
  const sheet = e.range.getSheet();
  if (sheet.getName() !== 'QA知識庫') return;

  const row = e.range.getRow();
  if (row <= 1) return;

  const apiKey = getGeminiApiKey();
  const qText = String(sheet.getRange(row, 4).getValue() || '').trim();
  const replyText = String(sheet.getRange(row, 7).getValue() || '').trim();
  const annotationText = String(sheet.getRange(row, 10).getValue() || '').trim();

  try {
    if (qText) {
      const qVec = getGeminiEmbedding(qText, apiKey);
      if (qVec) {
        const fileObj = saveVectorToJsonDrive(`QA_${row}_Q`, { type: 'question', text: qText, embedding: qVec });
        sheet.getRange(row, 11).setFormula(`=HYPERLINK("${fileObj.url}", "檢視問題向量JSON")`);
      }
    }
    if (replyText) {
      const rVec = getGeminiEmbedding(replyText, apiKey);
      if (rVec) {
        const fileObj = saveVectorToJsonDrive(`QA_${row}_R`, { type: 'reply', text: replyText, embedding: rVec });
        sheet.getRange(row, 12).setFormula(`=HYPERLINK("${fileObj.url}", "檢視回覆向量JSON")`);
      }
    }
    if (annotationText) {
      const fVec = getGeminiEmbedding(annotationText, apiKey);
      if (fVec) {
        const fileObj = saveVectorToJsonDrive(`QA_${row}_A`, { type: 'annotation', text: annotationText, embedding: fVec });
        sheet.getRange(row, 13).setFormula(`=HYPERLINK("${fileObj.url}", "檢視備註向量JSON")`);
      }
    }
  } catch (err) {
    Logger.log(`[handleQARowEdit Error] ${err.message}`);
  }

function sendBatchQuestionnaireFromCheckboxes() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('LINE對話');
  if (!sheet) {
    ss.toast("找不到 LINE對話 工作表！", "錯誤", 5);
    return;
  }

  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) {
    ss.toast("沒有可處理的病患訊息列。", "提示", 4);
    return;
  }

  let sentCount = 0;
  let failCount = 0;

  for (let row = 2; row <= lastRow; row++) {
    const checkboxVal = sheet.getRange(row, 2).getValue(); // Column B
    const userId = String(sheet.getRange(row, 3).getValue() || '').trim(); // Column C

    if (checkboxVal === true || checkboxVal === 'TRUE') {
      if (!userId) {
        sheet.getRange(row, 10).setValue('⚠ 無 User ID，無法發送');
        sheet.getRange(row, 2).setValue(false);
        failCount++;
        continue;
      }

      try {
        const success = sendDementiaScreeningFlexMessage(userId, 'Central');
        if (success) {
          sheet.getRange(row, 10).setValue('✅ 已批次發送問卷: ' + new Date().toLocaleTimeString());
          sentCount++;
        } else {
          sheet.getRange(row, 10).setValue('❌ 發送失敗');
          failCount++;
        }
      } catch (err) {
        sheet.getRange(row, 10).setValue('❌ 錯誤: ' + err.message);
        failCount++;
      } finally {
        // reset checkbox after processing
        sheet.getRange(row, 2).setValue(false);
      }
    }
  }

  ss.toast(`批次發送完成：成功 ${sentCount} 筆，失敗 ${failCount} 筆。`, "完成", 5);
}

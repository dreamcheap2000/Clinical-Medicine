function sendDementiaScreeningFlexMessage(userId, routeType = 'Central') {
  const cleanUserId = String(userId || '').trim();
  const scriptProps = PropertiesService.getScriptProperties();
  const channelAccessToken = getLineAccessToken(); // Relies on Webhook.gs
  
  let baseUrl = scriptProps.getProperty('GOOGLE_FORM_BASE_URL') || scriptProps.getProperty('GOOGLE_FORM_URL');
  if (!baseUrl || !baseUrl.startsWith('http')) {
    baseUrl = 'https://docs.google.com/forms/d/e/1FAIpQLScsPwKUHzWEOwCZ65nlyMD5UBgFZWZ9BXNj6uM3vIM2FHI0Cw/viewform';
  }
  baseUrl = baseUrl.split('?')[0];

  // Dynamically appends the LINE ID from Column C into the form's LINE ID entry field
  // NOTE: Replace 172749060 with your verified form entry ID if necessary
  const googleFormUrl = `${baseUrl}?usp=pp_url&entry.1626829425=${encodeURIComponent(cleanUserId)}`;
  
  const isCentral = (routeType === 'Central');
  const altTitle = isCentral ? '【核心循證】臨床失智評估量表 (CDR®)' : '【快速點擊】每日認知健康打卡';
  const headerColor = isCentral ? '#06C755' : '#FF8C00';
  const btnLabel = isCentral ? '填寫 CDR® 臨床評估表單' : '立即快速打卡';
  
  const payload = {
    to: cleanUserId,
    messages: [
      {
        type: 'flex',
        altText: altTitle,
        contents: {
          type: 'bubble',
          size: 'kilo',
          body: {
            type: 'box',
            layout: 'vertical',
            spacing: 'sm',
            contents: [
              { type: 'text', text: isCentral ? '📄 Central Route (CDR EBM)' : '⭐ Peripheral Route (Quick Check)', weight: 'bold', size: 'xs', color: headerColor },
              { type: 'text', text: altTitle, weight: 'bold', size: 'md', wrap: true, color: '#111111' },
              { type: 'text', text: isCentral ? '點擊下方按鈕填寫臨床失智評估量表 (CDR®)。系統已自動帶入您的識別編號。' : '輕鬆點選狀態，完成健康打卡。', size: 'xs', color: '#666666', wrap: true }
            ]
          },
          footer: {
            type: 'box',
            layout: 'vertical',
            spacing: 'sm',
            contents: [
              {
                type: 'button',
                style: 'primary',
                color: headerColor,
                height: 'sm',
                action: { type: 'uri', label: btnLabel, uri: googleFormUrl }
              }
            ]
          }
        }
      }
    ]
  };
  
  const response = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + channelAccessToken },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });
  
  return response.getResponseCode() === 200;
}

function handleCheckboxPushTrigger(e) {
  if (!e || !e.range) return;
  const range = e.range;
  const sheet = range.getSheet();
  
  if (sheet.getName() !== 'LINE對話' && sheet.getName() !== '工作表1') return;
  
  const row = range.getRow();
  const col = range.getColumn();
  const TRIGGER_COLUMN = 2; // Column B
  
  if (row > 1 && col === TRIGGER_COLUMN) {
    const cellValue = range.getValue();
    if (cellValue === true || cellValue === 'TRUE') {
      const ss = sheet.getParent();
      const userId = String(sheet.getRange(row, 3).getValue()).trim(); // Column C
      
      if (!userId) {
        ss.toast('⚠ 此列找不到 User ID，無法發送！', '發送失敗', 4);
        range.setValue(false);
        return;
      }
      
      try {
        const success = sendDementiaScreeningFlexMessage(userId, 'Central');
        if (success) {
          ss.toast(`✅ 已成功發送 CDR® 臨床評估表單給使用者 (${userId})！`, '發送成功', 4);
          sheet.getRange(row, 10).setValue('已發送問卷: ' + new Date().toLocaleTimeString());
        } else {
          ss.toast('❌ 發送失敗，請檢查 LINE API Token 或網址設定。', '錯誤', 5);
        }
      } catch (err) {
        ss.toast('❌ 執行錯誤: ' + err.message, '錯誤', 5);
      } finally {
        range.setValue(false); 
      }
    }
  }
}
